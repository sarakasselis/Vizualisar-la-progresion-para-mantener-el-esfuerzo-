## Código Arduino - Sesión 04

```cpp
#include <Adafruit_NeoPixel.h>

#define PULSE_PIN A0

#define LED_PIN_1 6
#define LED_PIN_2 7

#define NUM_LEDS 15

Adafruit_NeoPixel strip1(NUM_LEDS, LED_PIN_1, NEO_GRB + NEO_KHZ800);
Adafruit_NeoPixel strip2(NUM_LEDS, LED_PIN_2, NEO_GRB + NEO_KHZ800);

// Ajustar según las lecturas reales del sensor
int threshold = 550;

bool pulseDetected = false;

unsigned long lastBeatTime = 0;
unsigned long currentBeatTime = 0;

int bpm = 0;

void setup() {

  Serial.begin(9600);

  strip1.begin();
  strip2.begin();

  strip1.setBrightness(80);
  strip2.setBrightness(80);

  strip1.show();
  strip2.show();

  Serial.println("Sensor de pulso + 2 tiras LED");
  Serial.println("ValorSensor | BPM");
}

void loop() {

  int sensorValue = analogRead(PULSE_PIN);

  // Mostrar señal cruda y BPM en monitor serial
  Serial.print("Sensor: ");
  Serial.print(sensorValue);

  Serial.print("   BPM: ");
  Serial.println(bpm);

  // Detectar pulso
  if (sensorValue > threshold && !pulseDetected) {

    pulseDetected = true;

    currentBeatTime = millis();

    if (lastBeatTime > 0) {

      unsigned long intervalo = currentBeatTime - lastBeatTime;

      // Evita pulsos falsos demasiado rápidos o lentos
      if (intervalo > 300 && intervalo < 2000) {

        bpm = 60000 / intervalo;

        Serial.print(">>> LATIDO DETECTADO - BPM: ");
        Serial.println(bpm);

        actualizarColor(bpm);
      }
    }

    lastBeatTime = currentBeatTime;
  }

  // Esperar a que la señal vuelva a bajar
  if (sensorValue < threshold - 20) {
    pulseDetected = false;
  }

  delay(5);
}

void actualizarColor(int bpm) {

  if (bpm < 70) {

    // VERDE
    setColorAmbasTiras(0, 255, 0);

  } else if (bpm <= 100) {

    // AMARILLO
    setColorAmbasTiras(255, 150, 0);

  } else {

    // ROJO
    setColorAmbasTiras(255, 0, 0);
  }
}

void setColorAmbasTiras(int r, int g, int b) {

  for (int i = 0; i < NUM_LEDS; i++) {

    strip1.setPixelColor(i, strip1.Color(r, g, b));
    strip2.setPixelColor(i, strip2.Color(r, g, b));
  }

  strip1.show();
  strip2.show();
}
